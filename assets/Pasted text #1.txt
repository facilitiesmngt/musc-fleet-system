4. Data Model (Cosmos DB — single
container, two record types)

Trip record (type: &quot;trip&quot;):

id, type, vehicleTag, status
(&quot;in-progress&quot; | &quot;completed&quot;),

driverName, beginningOdometer,
endingOdometer,

from, to, jobFunction, passengers,
fuelLevel,

extendedTrip (bool), startedAt,
endedAt, signature (base64 image)

Issue record (type: &quot;issue&quot;):

id, type, vehicleTag, category, note,
reportedBy,

status (&quot;open&quot; |
&quot;resolved&quot; | &quot;false_alarm&quot;), reportedAt







5. Business Rules Implemented So Far

·&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;
Mileage chaining: starting reading on a new trip auto-suggests the previous trip's ending
reading

·&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;
Two-step flow: Start Trip (no signature needed) → End Trip (signature + ending mileage
required)

·&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;
Local travel toggle: auto-fills From/To with &quot;Charleston &amp; Surrounding Area&quot;

·&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;
Stale trip detection: flags a trip open &gt;12 hrs (or &gt;7 days if &quot;Extended Trip&quot;
checked) as stale, visible to anyone who scans that vehicle

·&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;
Soft guardrail: warns (but allows override) if a driver tries to start a trip while they
still have another vehicle's trip open

·&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;
Issue reporting: shows as a warning banner on that vehicle until an admin resolves it
(admin resolve UI is a future phase)

·&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;
PDF generation: 

o&nbsp;&nbsp;
Pulls all completed trips for a vehicle + month

o&nbsp;&nbsp;
Auto-splits into multiple pages (14 trips/page) with correct &quot;Page X
of Y&quot; numbering

o&nbsp;&nbsp;
Drops the instructions page entirely

o&nbsp;&nbsp;
Auto-fills driver initials (from all drivers that month) into the perjury
certification box

o&nbsp;&nbsp;
Stamps the last driver's actual drawn signature into the Driver
Signature box

o&nbsp;&nbsp;
Leaves the Supervisor Signature field open/fillable (not
flattened) for manual sign-off

o&nbsp;&nbsp;
YEAR field uses the full 4-digit year

o&nbsp;&nbsp;
Agency Code defaults to a constant (⚠️ you still need to set your real agency code —
see below)

o&nbsp;&nbsp;
Idle Time / Down Time / Condition / Commuting Mileage are entered
manually by the supervisor at generation time (not auto-calculated)

⚠️ Still Needs Your Input

·&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;
Real Agency Code — currently a placeholder (AGENCY_CODE_HERE) in generatePdf.js and
records.html

·&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;
Real vehicle list — currently hardcoded to 3 sample vehicles (SC-1042, SC-2077, SC-3311)
in driver.html and records.html dropdowns

const { app } =

&nbsp;

app.http('getVehicleStatus', {

 
methods: ['GET'],

 
authLevel: 'anonymous',

 
route: 'getVehicleStatus',

 
handler: async (request, context) =&gt; {

   
try {

     
const vehicleTag = request.query.get('vehicleTag');

     
if (!vehicleTag) return { status: 400, jsonBody: { error: 'vehicleTag
required' } };

&nbsp;

     
const { resources: trips } = await container.items.query({

       
query: &quot;SELECT * FROM c WHERE c.vehicleTag = @tag AND c.type =
'trip' ORDER BY c.startedAt DESC&quot;,

       
parameters: [{ name: '@tag', value: vehicleTag }]

     
}, { partitionKey: vehicleTag }).fetchAll();

&nbsp;

     
const { resources: issues } = await container.items.query({

       
query: &quot;SELECT * FROM c WHERE c.vehicleTag = @tag AND c.type =
'issue' AND c.status = 'open'&quot;,

       
parameters: [{ name: '@tag', value: vehicleTag }]

     
}, { partitionKey: vehicleTag }).fetchAll();

&nbsp;

     
const openTrip = trips.find(t =&gt; t.status === 'in-progress');

     
const lastCompleted = trips.find(t =&gt; t.status === 'completed');

&nbsp;

     
let result = { openIssues: issues, lastEndingOdometer: lastCompleted ?
lastCompleted.endingOdometer : null };

&nbsp;

     
if (openTrip) {

       
const hoursOpen = (Date.now() - new Date(openTrip.startedAt).getTime())
/ 3600000;

       
const staleThreshold = openTrip.extendedTrip ? 168 : 12;

       
result.openTrip = { ...openTrip, stale: hoursOpen &gt; staleThreshold };

     
}

&nbsp;

     
return { status: 200, jsonBody: result };

   
} catch (err) {

     
context.error(err);

     
return { status: 500, jsonBody: { error: 'Failed to load vehicle status'
} };

   
}

 
}

});

return { status: 404, jsonBody: { error: 'No completed trips found for
that vehicle/month' } };

     
}

&nbsp;

     
const templateBytes = fs.readFileSync(templatePath);

     
const meta = {

       
vehicleTag,

       
year: year,

       
monthPadded: String(month).padStart(2, '0'),

       
agencyCode: agencyCode || DEFAULT_AGENCY_CODE,

       
startMileage: trips[0].beginningOdometer,

       
endMileage: trips[trips.length - 1].endingOdometer,

       
milesUsed: Number(trips[trips.length - 1].endingOdometer) -
Number(trips[0].beginningOdometer),

       
idleTime: idleTime || '',

       
downTime: downTime || '',

       
condition: condition || '',

       
commutingMileage: commutingMileage || ''

     
};

&nbsp;

     
const chunks = [];

     
for (let i = 0; i &lt; trips.length; i += 14) chunks.push(trips.slice(i,
i + 14));

     
const totalPages = chunks.length;

&nbsp;

     
const finalDoc = await PDFDocument.create();

     
const helv = await finalDoc.embedFont(StandardFonts.Helvetica);

&nbsp;

     
for (let i = 0; i &lt; chunks.length; i++) {

       
const chunkDoc = await buildLogPage(templateBytes, chunks[i], meta, i +
1, totalPages, helv);

       
const [copiedPage] = await finalDoc.copyPages(chunkDoc, [0]);

       
finalDoc.addPage(copiedPage);

     
}

&nbsp;

     
const pdfBytes = await finalDoc.save();

&nbsp;

     
return {

       
status: 200,

       
headers: {

          'Content-Type': 'application/pdf',

          'Content-Disposition': `attachment;
filename=&quot;${vehicleTag}_${year}-${meta.monthPadded}.pdf&quot;`

       
},

       
body: Buffer.from(pdfBytes)

     
};

   
} catch (err) {

     
context.error(err);

     
return { status: 500, jsonBody: { error: 'Failed to generate PDF: ' +
err.message } };

   
}

 
}

});
